def system_msg(cwi):
    A = cwi["A"]
    B = cwi["B"]
    V = cwi["unit_cell_volume"]

    atoms_frac = cwi["atoms_frac"]
    atoms_cart = cwi["atoms_cart"]

    mp_grid = cwi["mp_grid"]
    num_k = cwi["num_k"]

    msg = f"""

                                   ------
                                   SYSTEM
                                   ------

                            Lattice Vectors (Ang)
                a_1    {'{:>.6f}'.format(A[0][0])}   {'{:>.6f}'.format(A[0][1])}   {'{:>.6f}'.format(A[0][2])}
                a_2    {'{:>.6f}'.format(A[1][0])}   {'{:>.6f}'.format(A[1][1])}   {'{:>.6f}'.format(A[1][2])}
                a_3    {'{:>.6f}'.format(A[2][0])}   {'{:>.6f}'.format(A[2][1])}   {'{:>.6f}'.format(A[2][2])}

                Unit Cell Volume:      {'{:>.5f}'.format(V)}  (Ang^3)

                      Reciprocal-Space Vectors (Ang^-1)
                b_1    {'{:>.6f}'.format(B[0][0])}   {'{:>.6f}'.format(B[0][1])}   {'{:>.6f}'.format(B[0][2])}
                b_2    {'{:>.6f}'.format(B[1][0])}   {'{:>.6f}'.format(B[1][1])}   {'{:>.6f}'.format(B[1][2])}
                b_3    {'{:>.6f}'.format(B[2][0])}   {'{:>.6f}'.format(B[2][1])}   {'{:>.6f}'.format(B[2][2])}

    """

    msg += """
*----------------------------------------------------------------------------*
|   Site       Fractional Coordinate          Cartesian Coordinate (Ang)     |
+----------------------------------------------------------------------------+
    """

    msg += "".join(
        [
            f"""
  {X}   {i}   {'{:>.5f}'.format(vf[0])}   {'{:>.5f}'.format(vf[1])}   {'{:>.5f}'.format(vf[2])}    |    {'{:>.5f}'.format(vc[0])}   {'{:>.5f}'.format(vc[1])}   {'{:>.5f}'.format(vc[2])}
    """
            for ((X, i), vf), vc in zip(atoms_frac.items(), atoms_cart.values())
        ]
    )

    msg += """
*----------------------------------------------------------------------------*
"""

    msg += f"""

                                ------------
                                K-POINT GRID
                                ------------

              Grid size =  {mp_grid[0]} x {mp_grid[1]} x {mp_grid[2]}      Total points = {num_k}
"""

    return msg

def relative_outdir(cwi, key):
    """
    outdir or mp_outdir relative to the directory of seedname.cwin, as written to seedname.cwout and seedname.hdf5,
    so that the outputs do not contain the absolute path of the directory where they were created.

    Args:
        cwi (dict): input parameters with the absolute outdir/mp_outdir and cwin_dir.
        key (str): "outdir" or "mp_outdir".

    Returns:
        str: relative path (unchanged if cwin_dir is not known).
    """
    v = cwi[key]
    cwin_dir = cwi.get("cwin_dir")
    if cwin_dir is None or not os.path.isabs(v):
        return v

    return os.path.relpath(v, cwin_dir)

def cwin_msg(cwi):

    label_width = 35
    value_width = 36

    msg = """
                               --------------
                               CLOSESTWANNIER
                               --------------
    """

    msg += "\n*----------------------------------------------------------------------------*\n"

    for k in cwin_info.keys():
        if cwi[k] == None:
            continue
        if k == "ket_amn":
            msg += f"| {k:<{label_width}} : {"":>{value_width}} |\n"
            for ket in cwi[k]:
                msg += f"| {"":<{label_width}}   {str(ket):>{value_width}} |\n"
        elif k in ("outdir", "mp_outdir"):
            msg += f"| {k:<{label_width}} : {relative_outdir(cwi, k):>{value_width}} |\n"
        else:
            msg += f"| {k:<{label_width}} : {str(cwi[k]):>{value_width}} |\n"

    msg += "*----------------------------------------------------------------------------*"

    return msg
